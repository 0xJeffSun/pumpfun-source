from pathlib import Path
import hashlib,json,subprocess,sys

P=Path(__file__).resolve().parent;W=P.parent
label=sys.argv[1]if len(sys.argv)>1 else 'probe-01'
assert Path(label).name==label and label not in ['.','..']
out=P/label;out.mkdir()
compiler=W/'layout-lane-linux-target2-sysroot/bin/rustc'
source=P/'abi_probe.rs';sha=lambda p:hashlib.file_digest(p.open('rb'),'sha256').hexdigest()
cmd=[str(compiler),'--crate-name','mayhem_data_abi','--edition','2021','--crate-type','lib','--target','sbpf-solana-solana','-C','opt-level=3','-C','panic=abort','-C','overflow-checks=on','--emit=llvm-ir','-o',str(out/'abi_probe.ll'),str(source)]
record={'command':cmd,'compiler_sha256':sha(compiler),'source_sha256':sha(source),'target_inventory':json.loads((W/'layout-lane-linux-target2-inputs.json').read_bytes()),'scope':'A separate ABI probe; no target program compilation or output rewrite.'}
(out/'launch.json').write_text(json.dumps(record,indent=2)+'\n')
with(out/'stdout').open('wb')as stdout,(out/'stderr').open('wb')as stderr:r=subprocess.run(cmd,stdout=stdout,stderr=stderr)
result={'returncode':r.returncode,'source_unchanged':sha(source)==record['source_sha256'],'compiler_unchanged':sha(compiler)==record['compiler_sha256'],'stdout_sha256':sha(out/'stdout'),'stderr_sha256':sha(out/'stderr')}
if r.returncode==0:result['llvm_ir_sha256']=sha(out/'abi_probe.ll')
(out/'result.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result,indent=2))
assert r.returncode==0 and result['source_unchanged'] and result['compiler_unchanged']
