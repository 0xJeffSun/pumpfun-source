  private ensureAtaInstructions(
    payer: PublicKey,
    owner: PublicKey,
    mint: PublicKey,
    ata: PublicKey,
    ataExists: boolean,
    tokenProgram: PublicKey,
  ): TransactionInstruction[] {
    return ataExists
      ? []
      : [
          createAssociatedTokenAccountIdempotentInstruction(
            payer,
            ata,
            owner,
            mint,
            tokenProgram,
          ),
        ];
  }
