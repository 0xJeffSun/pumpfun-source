"""Offline target-only BuyV3 flag discriminator; all transactions normally signed.

Uses a captured signed-fixture state; only explicitly listed curve/global bytes
are changed in the local VM. No admin instruction, mainnet access or sigverify
bypass. Synthetic state is not a claim of live mode reachability.
"""
import argparse,base64,hashlib,json,struct,sys
from pathlib import Path
ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'scripts'))
import pump_trade_fixture as f
OUT=Path(__file__).resolve().parent
parser=argparse.ArgumentParser();parser.add_argument('--candidate',type=Path);parser.add_argument('--output',type=Path,default=OUT/'funded-comparison.json');args=parser.parse_args()
sha=lambda x:hashlib.sha256(x).hexdigest()
TARGET=ROOT/'evidence/original/pump/deployed.so'
assert sha(TARGET.read_bytes())=='7a19affaa2c6187e04a5fe0ea298cee81fa5eb8c24cfcc30d85f615e89f7b02d'
assert all(f.t.verify_with_results())
user=f.Keypair.from_seed(bytes([231])*32)
assert user.pubkey()==f.msg.account_keys[0]
program=f.Pubkey.from_string(f.x['program'])
curve=f.Pubkey.from_string(f.x['accounts']['curve'])
global_key=f.Pubkey.from_string(f.x['accounts']['global'])
initial={key:f.vm.get_account(key)for key in f.msg.account_keys}

def run(mode,whitelist,amount,reserve,trailing,binary):
    vm=f.LiteSVM().with_sigverify(True).with_blockhash_check(True).with_transaction_history(0)
    for name in ['pump','pump_fees']:
        cap=json.loads((ROOT/f'evidence/original/{name}/capture.json').read_text())
        vm.add_program_from_file(f.Pubkey.from_string(cap['program']),binary if name=='pump' else ROOT/f'evidence/original/{name}/deployed.so')
    for name in ['spl_token','spl_token_2022','associated_token']:
        cap=json.loads((ROOT/f'evidence/dependencies/{name}/capture.json').read_text())
        vm.add_program_from_file(f.Pubkey.from_string(cap['program']),ROOT/f'evidence/dependencies/{name}/deployed.so')
    for key,account in initial.items():
        if account is not None and not account.executable:vm.set_account(key,account)
    ac=initial[curve];data=bytearray(ac.data)
    data[48]=0;data[81]=mode;data[82]=0
    struct.pack_into('<Q',data,24,reserve)
    struct.pack_into('<Q',data,32,1_000_000_000)
    vm.set_account(curve,f.Account(max(ac.lamports,1_010_000_000),bytes(data),ac.owner))
    ac=initial[global_key];gd=bytearray(ac.data)
    if whitelist:gd[451:451+32]=bytes(user.pubkey())
    else:gd[451:451+32]=bytes(32)
    vm.set_account(global_key,f.Account(ac.lamports,bytes(gd),ac.owner))
    ix=f.ixs[-1]
    data=bytes.fromhex('07051dc4f5176550')+struct.pack('<QQ',amount,10**12)+bytes(trailing)
    trade=f.Instruction(program,data,list(ix.accounts))
    tx=f.Transaction.new_signed_with_payer([*f.ixs[:-1],trade],user.pubkey(),[user],vm.latest_blockhash())
    assert all(tx.verify_with_results())
    before={str(k):state(vm.get_account(k))for k in f.msg.account_keys if k!=program}
    result=vm.send_transaction(tx);failed=isinstance(result,f.FailedTransactionMetadata);meta=result.meta()if failed else result
    after={str(k):state(vm.get_account(k))for k in f.msg.account_keys if k!=program}
    return {'mode':mode,'whitelist_equal':whitelist,'requested':amount,'real_token_reserves':reserve,'bool_bytes':trailing,
        'executed_sha256':sha(binary.read_bytes()),'transaction_base64':base64.b64encode(bytes(tx)).decode(),
        'signatures_verified':list(tx.verify_with_results()),'error':str(result.err())if failed else None,
        'logs':meta.logs(),'inner':json.loads(meta.to_json())['inner_instructions'],
        'event_payloads':[l[14:]for l in meta.logs()if l.startswith('Program data: ')],
        'before':before,'after':after,'watched_accounts_rolled_back':all(before[k]==after[k] for k in before if k!=str(user.pubkey())),
        'payer_lamport_delta':after[str(user.pubkey())]['lamports']-before[str(user.pubkey())]['lamports']}

def state(ac):
    return None if ac is None else dict(lamports=ac.lamports,owner=str(ac.owner),data_base64=base64.b64encode(ac.data).decode())

rows=[]
for mode,whitelist in [(0,False),(0,True),(1,False)]:
    for amount,reserve in [(500_000,1_000_000),(2_000_000,1_000_000),(2_000_000,0)]:
        for trailing in [[],[0],[1],[2]]:
            target=run(mode,whitelist,amount,reserve,trailing,TARGET)
            candidate=run(mode,whitelist,amount,reserve,trailing,args.candidate)if args.candidate else None
            if candidate:assert candidate['transaction_base64']==target['transaction_base64']
            rows.append(dict(case={k:target[k]for k in ['mode','whitelist_equal','requested','real_token_reserves','bool_bytes']},target=target,candidate=candidate,different_fields=[k for k in ['error','before','after','inner','event_payloads']if candidate and target[k]!=candidate[k]]))
report={'scope':__doc__,'fixture':str((ROOT/'research/imported-lab/analysis/pump_fun/curve-fresh-trades-v3-signed-surfpool.json').relative_to(ROOT)),
    'fixture_sha256':sha((ROOT/'research/imported-lab/analysis/pump_fun/curve-fresh-trades-v3-signed-surfpool.json').read_bytes()),
    'target_sha256':sha(TARGET.read_bytes()),'script_sha256':sha(Path(__file__).read_bytes()),'normal_signature_verification':True,
    'locally_mutated_offsets':{'curve':{'24':'real_token_reserves=per-case','32':'real_quote_reserves=1000000000','48':'complete=false','81':'is_mayhem_mode={mode}','82':'cashback=false'},'global':{'451':'whitelist_pda=user or zero; wire offset from Global Borsh IDL, decoded struct offset is0x288'}},
    'candidate_executed':args.candidate is not None,'candidate_sha256':sha(args.candidate.read_bytes())if args.candidate else None,'rows':rows}
args.output.write_text(json.dumps(report,indent=2)+'\n')
for r in rows:print(json.dumps(dict(**r['case'],target_error=r['target']['error'],candidate_error=r['candidate']['error']if r['candidate']else None,different_fields=r['different_fields'])))
