    // Local sBPF adapter: apply declared loader call relocations only in ephemeral
    // analysis memory. Original files and compiler inputs are never modified.
    private void applySbfCallRelocations() throws Exception {
        if (!currentProgram.getLanguageID().toString().equals("sBPF:LE:64:v1")) return;
        var memory = currentProgram.getMemory();
        var rel = memory.getBlock(".rel.dyn");
        var sym = memory.getBlock(".dynsym");
        var str = memory.getBlock(".dynstr");
        if (rel == null || sym == null || str == null) throw new IllegalArgumentException("sBPF relocation sections missing");
        var symbols = currentProgram.getSymbolTable();
        var syscallSymbols = new java.util.HashMap<String, Long>();
        var iterator = symbols.getAllSymbols(true);
        while (iterator.hasNext()) {
            var symbol = iterator.next();
            if (symbol.getAddress().getAddressSpace().getName().equals("syscall"))
                syscallSymbols.put(symbol.getName(), symbol.getAddress().getOffset());
        }
        int applied = 0;
        var changes = new StringBuilder();
        for (long offset = 0; offset < rel.getSize(); offset += 16) {
            var rec = rel.getStart().add(offset);
            long target = memory.getLong(rec);
            long info = memory.getLong(rec.add(8));
            if ((info & 0xffffffffL) != 10) continue;
            long index = info >>> 32;
            if (index >= sym.getSize() / 24) throw new IllegalArgumentException("bad sBPF symbol index");
            var entry = sym.getStart().add(index * 24);
            long nameOffset = Integer.toUnsignedLong(memory.getInt(entry));
            if (nameOffset >= str.getSize()) throw new IllegalArgumentException("bad sBPF symbol name");
            var name = new StringBuilder();
            for (long k = nameOffset; k < str.getSize(); k++) {
                byte ch = memory.getByte(str.getStart().add(k));
                if (ch == 0) break;
                if (name.length() >= 200) throw new IllegalArgumentException("long sBPF symbol");
                name.append((char)(ch & 255));
            }
            Long hash = syscallSymbols.get(name.toString());
            boolean local = Short.toUnsignedInt(memory.getShort(entry.add(6))) != 0;
            if (!local && hash == null) throw new IllegalArgumentException("unknown sBPF syscall: " + name);
            var address = currentProgram.getAddressFactory().getDefaultAddressSpace().getAddress(target);
            byte[] before = new byte[8]; memory.getBytes(address, before);
            if ((before[0] & 255) != 0x85 || before[2] != 0 || before[3] != 0)
                throw new IllegalArgumentException("sBPF call relocation is not on a CALL");
            currentProgram.getFunctionManager().removeFunction(address);
            clearListing(address, address.add(7));
            long immediate = local ? (memory.getLong(entry.add(8)) - target - 8) / 8 : hash;
            if (local && (memory.getLong(entry.add(8)) - target - 8) % 8 != 0)
                throw new IllegalArgumentException("unaligned local sBPF target");
            byte[] after = new byte[] {(byte)0x85, (byte)(local ? 0x10 : 0), 0, 0,
                (byte)immediate, (byte)(immediate >>> 8), (byte)(immediate >>> 16), (byte)(immediate >>> 24)};
            memory.setBytes(address, after);
            changes.append(Long.toHexString(target)).append(":").append(name).append(":")
                .append(HexFormat.of().formatHex(before)).append("->")
                .append(HexFormat.of().formatHex(after)).append(";");
            applied++;
        }
        currentProgram.getOptions(Program.PROGRAM_INFO).setString("REA local sBPF call relocations",changes.toString());
        println("REA local sBPF call relocations: " + applied + "; " + changes);
    }
